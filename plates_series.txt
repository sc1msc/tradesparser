# Допустимые буквы на российских госномерах
LETTERS = "АВЕКМНОРСТУХ"

# ---------------------------------------------------------
# ОСОБЫЕ СЕРИИ
# ---------------------------------------------------------
#
# Формат номера:
# А123ВС77
#
# * в исходном списке означает:
#   * после первой буквы — цифра
#   * в блоке букв — буква
#   * в регионе — цифра
#
# Для *** после региона допускаем как 2, так и 3 цифры,
# поскольку российские регионы бывают двух- и трёхзначными.
#

SPECIAL_SERIES = {
    "х***кх77":   r"^Х\d{3}КХ77$",
    "а***оо77":   r"^А\d{3}ОО77$",
    "в***оо77":   r"^В\d{3}ОО77$",
    "с***оо77":   r"^С\d{3}ОО77$",
    "м***оо77":   r"^М\d{3}ОО77$",

    "а***мо**":   r"^А\d{3}МО\d{2}$",
    "а***мо***":  r"^А\d{3}МО\d{2,3}$",

    "а***кр177":  r"^А\d{3}КР177$",
    "в***кр177":  r"^В\d{3}КР177$",

    "с***кр**":   r"^С\d{3}КР\d{2}$",
    "с***кр1**":  r"^С\d{3}КР1\d{2}$",

    "е***ре177":  r"^Е\d{3}РЕ177$",

    "****мр77":   rf"^[{LETTERS}]\d{{3}}МР77$",

    "а***мм**":   r"^А\d{3}ММ\d{2}$",
    # трёхзначные московские регионы (двузначные 77/97/99 ловит строка выше)
    "а***мм***москва": r"^А\d{3}ММ(?:177|197|199|777|797|799|977)$",

    "а***мр**":   r"^А\d{3}МР\d{2}$",
    "а***мр***":  r"^А\d{3}МР\d{2,3}$",

    "м***оо77":   r"^М\d{3}ОО77$",
}


# ---------------------------------------------------------
# КРАСИВЫЕ НОМЕРА
# ---------------------------------------------------------

BEAUTIFUL_PATTERNS = {

    # А111ВС77
    # Три одинаковые цифры
    "три одинаковые цифры":
        rf"^[{LETTERS}](\d)\1\1[{LETTERS}]{{2}}\d{{2,3}}$",

    # А123АА77
    # Три одинаковые буквы с учётом первой буквы:
    # А ... А А
    #
    # Например:
    # А123АА77
    # М777ММ99
    "три одинаковые буквы":
        rf"^([{LETTERS}])\d{{3}}\1\1\d{{2,3}}$",

    # А121ВС77
    # Зеркальные цифры: 121, 232, 505, 777 и т. п.
    "зеркальные цифры":
        rf"^[{LETTERS}](\d)\d\1[{LETTERS}]{{2}}\d{{2,3}}$",

    # А123ВА77
    # Зеркальные буквы относительно первой буквы:
    # А ... В А
    "зеркальные буквы":
        rf"^([{LETTERS}])\d{{3}}([{LETTERS}])\1\d{{2,3}}$",

    # Регионы 77 и 99
    "старая Москва":
        rf"^[{LETTERS}]\d{{3}}[{LETTERS}]{{2}}(?:77|99)$",

    # А001ВС77 ... А009ВС77
    # Цифры от 001 до 009
    "цифры 001-009":
        rf"^[{LETTERS}]00[1-9][{LETTERS}]{{2}}\d{{2,3}}$",

    # А100ВС77, А200ВС77 ... А900ВС77
    # Цифры, кратные 100
    "кратные 100":
        rf"^[{LETTERS}][1-9]00[{LETTERS}]{{2}}\d{{2,3}}$",
}
